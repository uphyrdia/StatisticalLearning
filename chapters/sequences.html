We begin with convergence of real sequences. Proofs are hidden until you open them.

\section{Convergence}\label{seq-convergence} \begin{def}\label{def-convergence}

A sequence \((x_n)_{n\geq1}\) converges to \(x\in\RR\) if, for every \(\varepsilon>0\), there exists \(N\in\NN\) such that \[ n\geq N \implies |x_n-x|\lt\varepsilon. \]

\end{def} \section{Properties of convergent sequences}\label{seq-properties} \subsection{Boundedness}\label{seq-boundedness} \begin{thm}\label{thm-bounded}

Every convergent real sequence is bounded.

\end{thm} \begin{proof}\label{proof-bounded}

Suppose \(x_n\to x\). By , choose \(N\geq1\) such that \(|x_n-x|\lt1\) whenever \(n\geq N\).

For those indices, the triangle inequality gives \[\label{bound-step} |x_n|\leq |x_n-x|+|x|\lt 1+|x|. \]

Thus the entire sequence is bounded by \[ M=\max\bigl(\{|x_k|:1\leq k\leq N\}\cup\{1+|x|\}\bigr). \]

\end{proof} \subsection{Uniqueness of the limit}\label{seq-uniqueness} \begin{thm}\label{thm-unique-limit}

A real sequence has at most one limit.

\end{thm} \begin{proof}

If \(x_n\to a\) and \(x_n\to b\), then \[ 0\leq |a-b|\leq |a-x_n|+|x_n-b|\longrightarrow 0. \]

Therefore \(a=b\).

\end{proof}
